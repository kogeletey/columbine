* Basic goals, Proxima(2025.06) version

- Integrate models output from openrouter
- Support MCP server for search and MCP servers in general
  - [[https://github.com/spences10/mcp-duckduckgo-search][MCP duckduckgo server]]
- Support Chrome File System Api for data storage, and its integration with MCP
  - [[https://developer.chrome.com/docs/capabilities/web-apis/file-system-access][The file system api]]
- Support document storage in localstorage
- Specialized editors, for example make the settings a JSON editor

Main setting in a json format:

#+BEGIN_SRC typescript

class SaveTypes {
    static browser = 'browser'
    // For chromium* browser - save is a direct, but on Firefox it downloads as an archive
    static fileSystem = 'file-system'
}

class BufferLayouts {
    static editor = '@core/editor'
    static panel = '@core/panel'
    static tree = '@core/file-tree'
}

class BufferAligments = {
    static left = 'left'
    static center = 'center'
    static right = 'right'
    static justify = 'justify'
}

class BufferPosition = {
    static top = 'top'
    static center = 'center'
    static bottom = 'bottom'
    static modal = 'modal'
}

class BufferOrientation = {
    static vertical = 'vertical'
    static horizontal = 'horizontal'
}

export type BlockId = string;

export type OutputBlocksApi<Type, Data> = {
   id?: BlockId;
   type: Type;
   data: Data;
}

export type LLM = {
    name: string;
    url: string;
    apiKey: string;
}

type JupyterSetting = {
  control_port: number;
  shell_port: number;
  transport: 'tcp' | 'ipc';
  signature_scheme: 'hmac-sha256';
  stdin_port: number;
  hb_port: number;
  ip: string;
  iopub_port: number;
  key: string;
}

type MCPServer = {
  command?: string;
  args?: string[];
  transport?: 'sse' | 'ws';
  url?: string;
  env?: Record<string, string>;
}

type FilePath = string;

export type Settings = {
    // optional one file or separate json files, with a filepath to kernel or mcp
    kernel?: JupyterSetting | FilePath
    mcp?: Record<string,MCPServer> | FilePath
    llms?: LLM[]
    layouts?: BufferProperties,
    imports?: Record<string,string>
    debug?: boolean;
    save: {
        auto?: boolean
    }
}

export type BufferProperty = {
    id: BlockId
    extend?: string;
    layout: BufferLayouts
    aligment: BufferAligments
    position: BufferPosition
    orientation: BufferOrientation
    pinBuffers?: BlockId[]
}

export type BufferProperties = BufferProperty[]
#+END_SRC
